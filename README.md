## **1. To-Do List API**



```bash
const express = require("express");
const app = express();
app.use(express.json());

let tasks = [];

// Add a task
app.post("/tasks", (req, res) => {
  const { task } = req.body;
  tasks.push({ id: tasks.length + 1, task, completed: false });
  res.status(201).json({ message: "Task added", tasks });
});

// Fetch all tasks
app.get("/tasks", (req, res) => {
  res.json(tasks);
});

// Mark a task as completed
app.put("/tasks/:id", (req, res) => {
  const { id } = req.params;
  const task = tasks.find((t) => t.id === parseInt(id));
  if (task) {
    task.completed = true;
    res.json({ message: "Task marked as completed", task });
  } else {
    res.status(404).json({ message: "Task not found" });
  }
});

// Delete a task
app.delete("/tasks/:id", (req, res) => {
  const { id } = req.params;
  tasks = tasks.filter((t) => t.id !== parseInt(id));
  res.json({ message: "Task deleted", tasks });
});

app.listen(3000, () => console.log("Server running on http://localhost:3000"));

```
## **2. Weather App**



```bash
const express = require("express");
const axios = require("axios");
const app = express();

app.get("/weather", async (req, res) => {
  const { city } = req.query;
  const apiKey = "YOUR_OPENWEATHERMAP_API_KEY";
  const url = `https://api.openweathermap.org/data/2.5/weather?q=${city}&appid=${apiKey}`;

  try {
    const response = await axios.get(url);
    const { temp, humidity } = response.data.main;
    const weather = response.data.weather[0].description;
    res.json({ city, temp, humidity, weather });
  } catch (error) {
    res.status(500).json({ message: "Error fetching weather data" });
  }
});

app.listen(3000, () => console.log("Server running on http://localhost:3000"));
```

## **3. URL Shortener**



```bash
const express = require("express");
const app = express();
app.use(express.json());

const urlDatabase = {};

// Shorten a URL
app.post("/shorten", (req, res) => {
  const { longUrl } = req.body;
  const shortCode = Math.random().toString(36).substring(2, 7);
  urlDatabase[shortCode] = longUrl;
  res.json({ shortUrl: `http://localhost:3000/${shortCode}` });
});

// Redirect to the original URL
app.get("/:shortCode", (req, res) => {
  const { shortCode } = req.params;
  const longUrl = urlDatabase[shortCode];
  if (longUrl) {
    res.redirect(longUrl);
  } else {
    res.status(404).json({ message: "URL not found" });
  }
});

app.listen(3000, () => console.log("Server running on http://localhost:3000"));
```

## **7. Currency Converter**



```bash
const express = require("express");
const axios = require("axios");
const app = express();

app.get("/convert", async (req, res) => {
  const { from, to, amount } = req.query;
  const apiKey = "YOUR_EXCHANGERATESAPI_KEY";
  const url = `https://api.exchangeratesapi.io/latest?base=${from}&symbols=${to}`;
```

## **8. Authentication API**



```bash
const express = require("express");
const jwt = require("jsonwebtoken");
const app = express();
app.use(express.json());

const users = [];

// Register a user
app.post("/register", (req, res) => {
  const { username, password } = req.body;
  users.push({ username, password });
  res.status(201).json({ message: "User registered" });
});

// Login a user
app.post("/login", (req, res) => {
  const { username, password } = req.body;
  const user = users.find((u) => u.username === username && u.password === password);
  if (user) {
    const token = jwt.sign({ username }, "secret_key");
    res.json({ token });
  } else {
    res.status(401).json({ message: "Invalid credentials" });
  }
});

app.listen(3000, () => console.log("Server running on http://localhost:3000"));
```

## **17. Simple Calculator API**



```bash
const express = require('express');
const app = express();

app.get('/calculate', (req, res) => {
    const { operation, a, b } = req.query;
    let result;
    switch (operation) {
        case 'add':
            result = parseFloat(a) + parseFloat(b);
            break;
        case 'subtract':
            result = parseFloat(a) - parseFloat(b);
            break;
        case 'multiply':
            result = parseFloat(a) * parseFloat(b);
            break;
        case 'divide':
            result = parseFloat(a) / parseFloat(b);
            break;
        default:
            return res.status(400).json({ message: 'Invalid operation' });
    }
    res.json({ result });
});

app.listen(3000, () => console.log('Server running on http://localhost:3000'));
```
